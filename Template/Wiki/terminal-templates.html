2.c PowerShell - fond bleu

▮ Start-MyService.ps1 - PowerShell—   ☐   ✕
param(
    [Parameter(Mandatory)][string]$ServiceName,
    [int]$TimeoutSec = 30
)
$ErrorActionPreference = 'Stop'

try {
    $svc = Get-Service -Name $ServiceName
    if ($svc.Status -ne 'Running') {
        Start-Service -Name $ServiceName
        Write-Host "[INFO] $ServiceName demarre" -ForegroundColor Green
    }
}
catch {
    Write-Error "Echec : $($_.Exception.Message)"
    exit 1
}

2.d Groovy / Jenkinsfile - Dracula

● ● ●   Jenkinsfile - Groovy
pipeline {
    agent { label 'linux' }
    options { timestamps(); timeout(time: 30, unit: 'MINUTES') }
    environment { APP_ENV = 'dev' }

    stages {
        stage('Build') {
            steps { sh './gradlew clean build' }
        }
        stage('Deploy') {
            when { branch 'main' }
            steps {
                withCredentials([string(credentialsId: 'deploy-token', variable: 'TOKEN')]) {
                    sh "./deploy.sh ${env.APP_ENV}"
                }
            }
        }
    }
    post {
        success { echo 'OK' }
        failure { echo 'Pipeline en echec' }
    }
}

2.e Python - Monokai

● ● ●   healthcheck.py - Python
#!/usr/bin/env python3
import sys
import requests


@retry(times=3)
def check(url: str, timeout: int = 5) -> bool:
    """Retourne True si le service repond 200."""
    try:
        r = requests.get(url, timeout=timeout)
        return r.status_code == 200
    except requests.RequestException as exc:
        print(f"[ERROR] {exc}", file=sys.stderr)
        return False


if __name__ == "__main__":
    sys.exit(0 if check("http://localhost:8080/health") else 1)

2.g Rust - Monokai

● ● ●   src/main.rs - Rust
use std::env;
use std::process;

#[derive(Debug)]
struct Config {
    name: String,
}

fn main() {
    let name = env::args().nth(1).unwrap_or_else(|| {
        eprintln!("Usage: app <name>");
        process::exit(1);
    });
    let cfg = Config { name };
    println!("Hello, {}! {:?}", cfg.name, cfg);
}
